Thank you for installing {{ .Chart.Name }} v{{ .Chart.AppVersion }}!

=== Access the application ===
{{- if .Values.ingress.enabled }}
  External URL (ALB): http://{{ .Values.ingress.host }}/
{{- else }}
  Port-forward for local access:
    kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "myapp.fullname" . }} 8080:{{ .Values.service.port }}
  Then open: http://localhost:8080
{{- end }}

=== Health checks ===
  Liveness : GET /healthz  (port: api-web / {{ .Values.containerPort }})
  Readiness: GET /readyz   (port: api-web / {{ .Values.containerPort }})

=== Metrics ===
  Prometheus endpoint: GET /metrics  (port: api-web / {{ .Values.containerPort }})
{{- if .Values.serviceMonitor.enabled }}
  ServiceMonitor is ENABLED — Prometheus Operator will scrape automatically.
{{- else }}
  ServiceMonitor is DISABLED — pod annotations provide fallback scrape config.
  Enable with: --set serviceMonitor.enabled=true
{{- end }}

=== Logs ===
  kubectl logs -n {{ .Release.Namespace }} \
    -l app.kubernetes.io/name={{ include "myapp.name" . }} --follow

=== Scale ===
{{- if .Values.autoscaling.enabled }}
  HPA active: {{ .Values.autoscaling.minReplicas }}-{{ .Values.autoscaling.maxReplicas }} replicas
  Target CPU: {{ .Values.autoscaling.targetCPUUtilizationPercentage }}%
{{- else }}
  Static replicas: {{ .Values.replicaCount }}
  Enable HPA: --set autoscaling.enabled=true
{{- end }}
